// VALIDATE on own field data (true holdout, never used in training).

function round2(x) { return ee.Number(x).multiply(100).round().divide(100); }
function round2Array(arr) { return ee.Array(arr).multiply(100).round().divide(100); }

function maskS2clouds(image) {
  var scl = image.select('SCL');
  return image.updateMask(scl.neq(3).and(scl.neq(8)).and(scl.neq(9)).and(scl.neq(10)));
}

// 2. PER-COUNTRY DRY-SEASON COMPOSITE (for spectral indices)
// ------------------------------------------------------------
function buildDryComposite(geom, monthStart, monthEnd) {
  return ee.ImageCollection('COPERNICUS/S2_SR_HARMONIZED')
    .filterBounds(geom)
    .filter(ee.Filter.calendarRange(2016, 2026, 'year'))
    .filter(ee.Filter.calendarRange(monthStart, monthEnd, 'month'))
    .filter(ee.Filter.lt('CLOUDY_PIXEL_PERCENTAGE', 20))
    .select(['B2','B3','B4','B5','B6','B7','B8','B8A','B11','B12','SCL'])
    .map(maskS2clouds)
    .median()
    .clip(geom);
}

var s2_dry_ET = buildDryComposite(Ethiopia.geometry(), 11, 2);
var s2_dry_KE = buildDryComposite(Kenya.geometry(), 1, 2);
var s2_dry = ee.ImageCollection([s2_dry_ET, s2_dry_KE]).mosaic();

var B2=s2_dry.select('B2'), B3=s2_dry.select('B3'), B4=s2_dry.select('B4'),
    B5=s2_dry.select('B5'), B7=s2_dry.select('B7'),
    B8=s2_dry.select('B8'), B11=s2_dry.select('B11'), B12=s2_dry.select('B12');

var NDVI  = B8.subtract(B4).divide(B8.add(B4)).rename('NDVI');
var VIRE  = B8.subtract(B7).divide(B8.add(B7)).rename('VIRE');
var TCARI = B5.subtract(B4).multiply(3)
  .subtract(B5.subtract(B3).multiply(0.6).multiply(B5.divide(B4))).rename('TCARI');
var DFI   = ee.Image(100).multiply(ee.Image(1).subtract(B12.divide(B8)))
  .multiply(B4.divide(B11)).rename('DFI');

// 3. PER-COUNTRY WET-SEASON COMPOSITE (for NDVI_amplitude)
// ------------------------------------------------------------
function buildWetComposite(geom, monthStart, monthEnd) {
  return ee.ImageCollection('COPERNICUS/S2_SR_HARMONIZED')
    .filterBounds(geom)
    .filter(ee.Filter.calendarRange(2016, 2026, 'year'))
    .filter(ee.Filter.calendarRange(monthStart, monthEnd, 'month'))
    .filter(ee.Filter.lt('CLOUDY_PIXEL_PERCENTAGE', 20))
    .select(['B4','B8','SCL'])
    .map(maskS2clouds)
    .median()
    .clip(geom);
}

var s2_wet_ET = buildWetComposite(Ethiopia.geometry(), 6, 9);
var s2_wet_KE = buildWetComposite(Kenya.geometry(), 3, 5);
var s2_wet = ee.ImageCollection([s2_wet_ET, s2_wet_KE]).mosaic();

var NDVI_wet = s2_wet.normalizedDifference(['B8','B4']).rename('NDVI_wet');
var NDVI_amplitude = NDVI_wet.subtract(NDVI).rename('NDVI_amplitude');

// OVERSAMPLING (calibration only)
// ------------------------------------------------------------
function balanceByOversampling(fc) {
  var c0 = fc.filter(ee.Filter.eq('label',0));
  var c1 = fc.filter(ee.Filter.eq('label',1));
  var n0 = c0.size(), n1 = c1.size(), nMax = n0.max(n1);
  var c0List = c0.toList(n0);
  var c0Over = ee.FeatureCollection(ee.List.sequence(0, nMax.subtract(1))
    .map(function(i){return ee.Feature(c0List.get(ee.Number(i).mod(n0)));}));
  var c1List = c1.toList(n1);
  var c1Over = ee.FeatureCollection(ee.List.sequence(0, nMax.subtract(1))
    .map(function(i){return ee.Feature(c1List.get(ee.Number(i).mod(n1)));}));
  return c0Over.merge(c1Over);
}
var calibrationBalanced = balanceByOversampling(calibrationTraining);

// already-fitted classifier silently fails, producing a scalar

  numberOfTrees: 100, minLeafPopulation: 2, seed: 42
}).setOutputMode('MULTIPROBABILITY').train({
  features: calibrationBalanced, classProperty: 'label', inputProperties: inputBands
});
